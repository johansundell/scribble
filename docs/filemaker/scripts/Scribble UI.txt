=== Scribble UI Save ===
# "Save" button on Scribble_Edit. Queues the edit in PendingEdits; never writes to Notes.
Set Error Capture [ On ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $body ; Value: ScribbleUI::gBody ]
Set Variable [ $noteID ; Value: ScribbleUI::gNoteID ]
Set Variable [ $baseRevision ; Value: ScribbleUI::gBaseRevision ]
If [ IsEmpty ( Scribble_TitleOf ( $body ) ) ]
    Show Custom Dialog [ "Scribble" ; "Type a title on the first line." ; Buttons: "OK" ]
    Go to Field [ ScribbleUI::gBody ]
    Exit Script [ ]
End If
If [ not IsEmpty ( $noteID ) and not IsEmpty ( Scribble_MyWaitingUpdate::ID ) ]
    # Fold the save into your waiting update and keep its BaseRevision, so a second save never conflicts with the first.
    Set Field [ Scribble_MyWaitingUpdate::Body ; $body ]
    Commit Records/Requests [ With dialog: Off ]
    Set Variable [ $error ; Value: Get ( LastError ) ]
    If [ $error ≠ 0 ]
        Revert Record/Request [ With dialog: Off ]
        Go to Layout [ "Scribble_Edit" ]
        Show Custom Dialog [ "Scribble" ; "Could not save the edit (FileMaker error " & $error & ")." ; Buttons: "OK" ]
        Exit Script [ ]
    End If
Else
    Go to Layout [ "Scribble_PendingEdits" ]
    New Record/Request
    Set Field [ PendingEdits::Kind ; If ( IsEmpty ( $noteID ) ; "create" ; "update" ) ]
    Set Field [ PendingEdits::NoteID ; $noteID ]
    Set Field [ PendingEdits::BaseRevision ; If ( IsEmpty ( $noteID ) ; "" ; $baseRevision ) ]
    Set Field [ PendingEdits::Body ; $body ]
    Commit Records/Requests [ With dialog: Off ]
    Set Variable [ $error ; Value: Get ( LastError ) ]
    If [ $error ≠ 0 ]
        Revert Record/Request [ With dialog: Off ]
        Go to Layout [ "Scribble_Edit" ]
        Show Custom Dialog [ "Scribble" ; "Could not save the edit (FileMaker error " & $error & ")." ; Buttons: "OK" ]
        Exit Script [ ]
    End If
End If
Perform Script [ "Scribble UI Show List" ]
Show Custom Dialog [ "Scribble" ; "Saved. It will appear in Scribble within a minute." ; Buttons: "OK" ]

=== Scribble UI Delete ===
# "Delete" button on Scribble_Edit. Queues a delete in PendingEdits.
Set Error Capture [ On ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $noteID ; Value: ScribbleUI::gNoteID ]
Set Variable [ $baseRevision ; Value: ScribbleUI::gBaseRevision ]
If [ IsEmpty ( $noteID ) ]
    # A new note that was never saved: nothing to delete.
    Perform Script [ "Scribble UI Show List" ]
    Exit Script [ ]
End If
Show Custom Dialog [ "Delete note" ; "Delete \"" & Scribble_EditNote::Title & "\"? This cannot be undone." ; Buttons: "Delete", "Cancel" ]
If [ Get ( LastMessageChoice ) ≠ 1 ]
    Exit Script [ ]
End If
Go to Layout [ "Scribble_PendingEdits" ]
New Record/Request
Set Field [ PendingEdits::Kind ; "delete" ]
Set Field [ PendingEdits::NoteID ; $noteID ]
Set Field [ PendingEdits::BaseRevision ; $baseRevision ]
Commit Records/Requests [ With dialog: Off ]
Set Variable [ $error ; Value: Get ( LastError ) ]
If [ $error ≠ 0 ]
    Revert Record/Request [ With dialog: Off ]
    Go to Layout [ "Scribble_Edit" ]
    Show Custom Dialog [ "Scribble" ; "Could not save the edit (FileMaker error " & $error & ")." ; Buttons: "OK" ]
    Exit Script [ ]
End If
# The delete replaces your waiting update of this note. Removed only now, so a failed save above loses nothing.
Enter Find Mode [ Pause: Off ]
Set Field [ PendingEdits::WaitingUpdateNoteID ; $noteID ]
Set Field [ PendingEdits::CreatedBy ; "==\"" & Get ( AccountName ) & "\"" ]
Perform Find [ ]
If [ Get ( FoundCount ) > 0 ]
    Delete All Records [ With dialog: Off ]
End If
Perform Script [ "Scribble UI Show List" ]
